# 成交量约束交易模拟器

## 问题描述

当前回测假设"日内任意时间可下单并成交"，但实际交易中：
1. 提交委托 ≠ 成交
2. 流动性不足时可能只部分成交
3. 需要撤单、改价、重试

## 方案1: 基于历史成交率的模拟

### 思路

使用历史数据统计每只股票在不同持仓规模下的**历史成交率**：

```python
def estimate_fill_rate(stock_code, target_shares, historical_data):
    """
    基于历史数据估算成交率
    
    historical_data: {
        'avg_daily_volume': 1000000,  # 日均成交量
        'volume_percentile_25': 800000,
        'volume_percentile_50': 1000000,
        'volume_percentile_75': 1200000,
    }
    
    Returns:
        fill_rate: 0.0 ~ 1.0
    """
    avg_vol = historical_data['avg_daily_volume']
    
    # 目标股数占日均成交量的比例
    participation_rate = target_shares / avg_vol
    
    # 成交率估算 (经验公式)
    if participation_rate <= 0.01:  # 1%以下
        return 0.95
    elif participation_rate <= 0.05:  # 5%以下
        return 0.80
    elif participation_rate <= 0.10:  # 10%以下
        return 0.60
    elif participation_rate <= 0.20:  # 20%以下
        return 0.40
    else:
        return 0.20  # 大单难以成交
```

## 方案2: 分批下单 + 追价策略

### 策略逻辑

```python
class SmartExecutor:
    """智能下单执行器"""
    
    def __init__(self, config):
        self.max_retry = config.get('max_retry', 3)
        self.price_improvement = config.get('price_improvement', 0.002)  # 每次改价提高0.2%
        self.wait_seconds = config.get('wait_seconds', 30)  # 等待30秒
        
    def execute_buy(self, stock, target_shares, limit_price):
        """
        执行买入
        
        策略:
        1. 先下一半的量在涨停价（几乎肯定成交）
        2. 剩下一半在目标价限价单
        3. 如果5分钟未成交，改价继续
        4. 直到成交或达到最大重试次数
        """
        filled = 0
        remaining = target_shares
        
        # 第一批: 激进单 (用更高的价格确保成交)
        aggressive_price = limit_price * (1 + self.price_improvement)
        filled += self._submit_and_wait('buy', stock, remaining // 2, aggressive_price)
        remaining = target_shares - filled
        
        # 第二批: 正常限价单
        if remaining > 0:
            filled += self._submit_and_wait('buy', stock, remaining, limit_price)
        
        return filled / target_shares  # 返回成交比例
    
    def execute_sell(self, stock, target_shares, limit_price):
        """
        执行卖出
        
        策略:
        1. 先下一半在跌停价（几乎肯定成交）
        2. 剩下一半在目标价
        3. 跌停时无法卖出，只能等打开
        """
        if self._is_limit_down(stock):
            return 0.0  # 跌停无法卖出
        
        filled = 0
        remaining = target_shares
        
        # 第一批: 激进单
        aggressive_price = limit_price * (1 - self.price_improvement)
        filled += self._submit_and_wait('sell', stock, remaining // 2, aggressive_price)
        remaining = target_shares - filled
        
        # 第二批: 正常限价单
        if remaining > 0:
            filled += self._submit_and_wait('sell', stock, remaining, limit_price)
        
        return filled / target_shares
    
    def _submit_and_wait(self, side, stock, shares, price):
        """提交订单并等待结果"""
        order_id = self._submit_order(side, stock, shares, price)
        
        for retry in range(self.max_retry):
            time.sleep(self.wait_seconds)
            
            status = self._check_order_status(order_id)
            
            if status == 'filled':
                return shares
            elif status == 'partial':
                return status['filled_shares']
            elif status == 'canceled':
                # 撤单后重新下单，价格微调
                new_price = self._adjust_price(price, side)
                order_id = self._submit_order(side, stock, shares, new_price)
            else:  # pending
                continue
        
        # 达到最大重试次数仍未成交
        self._cancel_order(order_id)
        return 0
```

def get_market_condition(date):
    """
    获取市场环境系数
    
    市场火热(涨停多): 成交率下降
    市场平淡: 成交率正常
    """
    # 可以用涨跌停股票数量、市场成交量等指标
    limit_up_count = get_limit_up_count(date)
    
    if limit_up_count > 100:  # 火热市场
        return 0.7  # 成交率打7折
    elif limit_up_count > 50:  # 偏热
        return 0.85
    else:  # 正常
        return 1.0

## 方案4: 真实数据驱动

### 使用Level2数据模拟

```python
def simulate_with_orderbook(orderbook_snapshot, side, shares, price):
    """
    使用Level2订单簿数据模拟成交
    
    orderbook_snapshot: {
        'asks': [(price, volume), ...],  # 卖一~卖N
        'bids': [(price, volume), ...],  # 买一~买N
    }
    """
    if side == 'buy':
        levels = orderbook_snapshot['asks']  # 买入看卖单
    else:
        levels = orderbook_snapshot['bids']  # 卖出看买单
    
    filled = 0
    cost = 0
    
    for level_price, level_volume in levels:
        if side == 'buy' and level_price > price:
            break  # 价格超过限价，不买
        if side == 'sell' and level_price < price:
            break  # 价格低于限价，不卖
        
        # 该档位能成交的数量
        can_fill = min(shares - filled, level_volume)
        filled += can_fill
        cost += can_fill * level_price
        
        if filled >= shares:
            break
    
    return {
        'filled': filled,
        'remaining': shares - filled,
        'avg_price': cost / filled if filled > 0 else None,
        'fill_rate': filled / shares
    }
```
